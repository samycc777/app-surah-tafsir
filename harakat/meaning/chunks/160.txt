6645	يَفْسُقُونَ (6:49)	الفسوق: العصيان والخروج عن حدود الشرع، والفسق: الخروج عن الطاعة. ومنه: فسقت الرُّطَبة: إذا خرجت من قشرها.
6646	أَقُولُ (6:50)	لا أقول: لا أدعي.
6647	خَزَائِنُ (6:50)	خزائن الله: مقدوراته ومرزوقاته التي استأثر بعلمها من شؤون خلقه.
6648	أَتَّبِعُ (6:50)	أنتهج.
6649	يُوحَى (6:50)	يوحى إلي: أبلغ بواسطة الوحي.
6650	يَسْتَوِي (6:50)	هل يستوي الطرفان: المراد "لا يتماثلان ولا يتعادلان".
6651	الْأَعْمَى (6:50)	فاقد البصيرة، والمراد: الكافر.
6652	وَالْبَصِيرُ (6:50)	البصير: صاحب البصيرة والمراد المؤمن.
6653	تَتَفَكَّرُونَ (6:50)	تتفكرون: تعملون عقولكم وتتدبرون.
6654	وَأَنْذِرْ (6:51)	وبلغ وأعلم.
6655	يُحْشَرُوا (6:51)	يجمعوا للحساب بعد البعث من القبور.
6656	دُونِهِ (6:51)	من دونه: غيره.
6657	شَفِيعٌ (6:51)	طالب تجاوز عن السيئة.
6658	تَطْرُدِ (6:52)	لا تطرد: لا تبعد.
6659	بِالْغَدَاةِ (6:52)	الغداة: أول النهار وهو الوقت ما بين الفجر إلى طلوع الشمس.
6660	يُرِيدُونَ (6:52)	يرغبون ويبتغون.
6661	وَجْهَهُ (6:52)	يريدون الله بطاعتهم، ويطلبون ثواب الله، وصفة الوجه ثابتة لله تعالى على ما يليق به.
6662	فَتَطْرُدَهُمْ (6:52)	فتبعدهم.
6663	الظَّالِمِينَ (6:52)	المتجاوزين للحد بوضع الشيء في غير موضعه، وأصل الظلم: مجاوزة الحد، ووضع الشيء غير موضعه ومنه يُقال من أشبه أباه فما ظلم، أي: ما وضع الشبه غير موضعه.
6664	فَتَنَّا (6:53)	فتنا بعضهم ببعض: ابتلينا بعض عبادنا ببعض بتباين حظوظهم من الأرزاق والأخلاق أو ابتليناهم وأوقعناهم في الفتنة.
6665	لِيَقُولُوا (6:53)	ليتكلموا.
6666	مَنَّ (6:53)	أنعم، والمنّةُ: النعمة الثقيلة، ويُقال ذلك من وجهين: أحدهما: أن يكون ذلك بالفعل، فيقال: منَّ فلان على فلان: إذا أثقله بالنّعمة، والثاني: أن يكون ذلك بالقول، وذلك مستقبح فيما بين الناس إلا عند كفران النّعمة.
6667	بِأَعْلَمَ (6:53)	بأكثر علما وإحاطة. والعلم: من صفات الله عز وجل.
6668	بِالشَّاكِرِينَ (6:53)	الشاكرين: الذاكرين لنعمة الله، المثنين عليه بها.
6669	بِآيَاتِنَا (6:54)	القرآن والحجج والبراهين.
6670	سَلَامٌ (6:54)	لفظ تحية وتسليم.
6671	كَتَبَ (6:54)	قضى وأوجب تفضلاً وإحساناً.
6672	سُوءًا (6:54)	قبحا، ويراد به الإثم والذنب.
6673	بِجَهَالَةٍ (6:54)	بطيش وسفاهة، وكل عاص مسيء جاهل.
6674	تَابَ (6:54)	رجع عن المعاصي.
6675	نُفَصِّلُ (6:55)	نبين ونوضح.
6676	الْآيَاتِ (6:55)	المعجزات والدلائل والعبر.
6677	وَلِتَسْتَبِينَ (6:55)	ولتظهر وتتضح.
6678	سَبِيلُ (6:55)	سبيل المجرمين: طريق أهل الباطل المخالفين للرسل.
6679	الْمُجْرِمِينَ (6:55)	جمع مجرم. والمُجْرِمُ هو الآثِم الذي ارتَكَبَ ما لا يحِلُّ ل. والإجرامُ معناه: الإثْم.
6680	نُهِيتُ (6:56)	نهيت أن أعبد: أمرت بعدم العبادة.
6681	أَعْبُدَ (6:56)	أنقاد وأخضع، والعبادة: الطاعة مقرونة بالتذلل والخضوع، ولا تليق إلا بمن له غاية الإفضال، وهو الباري تعالى، وهي اسم جامع لكل ما يحبه الله ويرضاه من الأعمال، والأقوال الظاهرة والباطنة.
6682	تَدْعُونَ (6:56)	تعبدون.
6683	أَتَّبِعُ (6:56)	لا أتبع: لا أنقاد.
6684	أَهْوَاءَكُمْ (6:56)	ما تهواه أنفسكم وتميل إليه.
6685	ضَلَلْتُ (6:56)	ضللت الطريق: لم أهتد إليه.
6686	الْمُهْتَدِينَ (6:56)	المستجيبين للهداية.
6687	بَيِّنَةٍ (6:57)	بصيرة واضحة من شريعة الله.
6688	وَكَذَّبْتُمْ (6:57)	كذبتم: لم تؤمنوا.
6689	تَسْتَعْجِلُونَ (6:57)	تتعجلون في الأمر وتطلبونه على وجه السرعة والمراد إنزال العذاب.
6690	الْحُكْمُ (6:57)	القضاء والفصل.
6691	يَقُصُّ (6:57)	يقص الحق: يرويه ويبينه بياناً شافياً.
6692	الْفَاصِلِينَ (6:57)	الحاكمين.
6693	لَقُضِيَ (6:58)	لقضي الأمر: لفصل فيه وحسمت المسألة.
6694	بِالظَّالِمِينَ (6:58)	الظالمين: الذين تجاوزوا حدهم فأشركوا معه غيره، أصل الظلم: مجاوزة الحد، ووضع الشيء غير موضعه ومنه يُقال من أشبه أباه فما ظلم، أي: ما وضع الشبه غير موضعه.
6695	مَفَاتِحُ (6:59)	مفاتح الغيب: خزائن الغيب، ومنها: علم الساعة، ونزول الغيث، وما في الأرحام، والكسب في المستقبل، ومكان موت الإنسان.
6696	لَا (6:59)	حرف نفي يفيد التوكيد.
