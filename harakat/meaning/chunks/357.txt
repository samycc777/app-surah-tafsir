20199	فَالْمُغِيرَاتِ (100:3)	قَسَمٌ من الله تعالى بالخيل الـمندفعات. و(الـمُغِيرَات) مُشْتَقَّةٌ من (الغَوْر)، وأصل (الغَوْر) في اللغة يدل على: وُرُودِ شيءٍ في شيءٍ وبُلُوغِهِ قَعْرَه ومُنخَفَضَه. ومن ذلك: غَوْرُ الخيلِ في داخلِ مُحيطِ العَدُوِّ وسَيْرُها إليه.
20200	صُبْحًا (100:3)	في أول النهار.
20201	فَأَثَرْنَ (100:4)	فهَيَّجْن ونَشَرْنَ.
20202	بِهِ (100:4)	بهذا الـجَرْي والعَدْوِ.
20203	نَقْعًا (100:4)	غبارًا منتشرًا.
20204	فَوَسَطْنَ (100:5)	فبَلَغْنَ ووَصَلْنَ إلى الداخِلِ والوَسط.
20205	بِهِ (100:5)	بفوارِسِهِنَّ ورُكَّابِهِنَّ.
20206	جَمْعًا (100:5)	جموع الأعداء.
20207	الْإِنْسَانَ (100:6)	ابن آدم، ذكرًا أو أنثى.
20208	لِرَبِّهِ (100:6)	لله تعالى. والرب هو المالك المربي المتصرف المدبر لخلقه كما يشاء على ما تقتضيه حكمته. والهاء ضميرٌ متصلٌ عائدٌ إلى الإنسان.
20209	لَكَنُودٌ (100:6)	لـمَنُوعٌ للخير الذي يريده منه ربُّه، جَحُودٌ لنِعَمِ الله تعالى. وأصل (كَنَدَ) في اللغة يدل على: حَبْسِ الشيءِ لـما في بطنه، فلا يُظهرُه ولا يُبْرِزُه. فالكَنُودُ رغم حصوله على النعمة إلا أنه يَكتُمُها ولا يُبدِيها بالشُّكرِ والتَّحَدُّث بها. ومنه أيضًا سُميت الأرض التي لا تُنبت شيئًا: (أرضٌ كَنُود)؛ لكونها لا تُنْبِتُ شيئًا.
20210	وَإِنَّهُ (100:7)	وإن الإنسان أيضًا. (إن): حرف توكيد ونصب يفيد تأكيد مضمون الجملة. والهاء ضميرٌ متصلٌ عائدٌ إلى الإنسان.
20211	ذَلِكَ (100:7)	أي: منعِه للخيرِ وجحودِه نعمَ ربِّه.
20212	لَشَهِيدٌ (100:7)	لـمُقِرٌّ مُعْتَرِفٌ.
20213	لِحُبِّ (100:8)	لـمَيْلِ نفسِه إليه.
20214	الْخَيْرِ (100:8)	الـمال.
20215	لَشَدِيدٌ (100:8)	لَقَوِيٌّ.
20216	يَعْلَمُ (100:9)	يتَذّكَّرُ ويُوقِنُ.
20217	بُعْثِرَ (100:9)	أُثِير واستُخرجَ ونُثِر.
20218	الْقُبُورِ (100:9)	جمع قبر، وقبر الإنسان: موضعُ دفنِه.
20219	وَحُصِّلَ (100:10)	وأُظْهِرَ الـمَسْتُورُ وجُمِعَ لأجل الحِسَاب.
20220	الصُّدُورِ (100:10)	الـمراد هنا: القلوب التي هي مَحَلُّ النِّيَّات وأعمال الباطن. والصدر من الإنسان: الجزء الممتد من أسفل العنق إلى فضاء الجوف، ويُطلق في القرآن على القلب لوجوده فيه.
20221	رَبَّهُمْ (100:11)	الله تعالى. والرب هو المالك المربي المتصرف المدبر لخلقه كما يشاء على ما تقتضيه حكمته. والهاء ضميرٌ متصلٌ عائدٌ إلى بني آدم.
20222	بِهِمْ (100:11)	بالناس. الباء: حرف جر يفيد معنى الإلصاق. والضمير المتصل عائدٌ إلى بني آدم.
20223	يَوْمَئِذٍ (100:11)	ذلك اليوم، وهو يوم القيامة.
20224	لَخَبِيرٌ (100:11)	خبير: صفةٌ لله سبحانه وتعالى، والخبير: هو المطلع على حقيقة الأشياء، فلا تخفى على الله خافية، وهو عالم بالكليات والجزئيات.
20225	الْقَارِعَةُ (101:1)	ساعة القيامة. وأصل القَرْع في اللغة: ضربُ شيءٍ بشيءٍ آخَر بِشِدَّة. وسُميت القيامَةُ قارعةً؛ لأنها تَـقْرَعُ القـلوبَ بأهوالها.
20226	يَوْمَ (101:4)	حينَ. والمعنى: تَكونُ القارعةُ وتحصُلُ في ذلك اليومِ الذي يكونُ الناسُ فيه على الأحوالِ التي سيأتي ذكرُها في بقية السورة.
20227	يَكُونُ (101:4)	يَصِيرُ.
20228	النَّاسُ (101:4)	بنو آدم. والناس: اسم للجمع من بني آدم، لا واحد له من لفظه.
20229	كَالْفَرَاشِ (101:4)	الكاف للتشبيه. والفراش: جنسٌ من الحشراتِ يَتَهافَتُ حولَ النارِ فيَحْتَرِقُ.
20230	الْمَبْثُوثِ (101:4)	الـمُنتشر الـمُتَفَرِّق.
20231	وَتَكُونُ (101:5)	وتَصِيرُ.
20232	الْجِبَالُ (101:5)	جمعُ (جَبَلْ)، وهو: ما ارتفع من الأرض إذا عَظُمَ وطَالَ.
20233	كَالْعِهْنِ (101:5)	كالصوف المصبوغ بألوان مختلفة. وأصل العِهْن في اللغة يدل على: اللين والاسترخاء والسُّهولة. ومنه سُمي الصُّوفُ عِهْنًا؛ لِلِينِهِ وسُهُولَتِه واسْتِرْخَائِه.
20234	الْمَنْفُوشِ (101:5)	الـمُفَرَّق بالأصابع الذي يُنْفَشُ باليد، فَيَصيرُ هباءً ويَزول. وأصل النَّفْشِ في اللغة يدل على: الانتشار بعد الانضمام، أو على تفريق أجزاء الشيء.
20235	مَنْ (101:6)	اسم موصول بمعنى (الذي) يختص بذوات من يعقل.
20236	ثَقُلَتْ (101:6)	رَجَحَتْ.
